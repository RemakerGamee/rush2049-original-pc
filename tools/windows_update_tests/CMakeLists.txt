cmake_minimum_required(VERSION 3.20)
project(RushUpdateTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(RUSH_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
enable_testing()
add_executable(RushUpdateModelCheck model-check.cpp)
target_include_directories(RushUpdateModelCheck PRIVATE "${RUSH_JSON_INCLUDE_DIR}")
add_test(NAME release_identity_model COMMAND RushUpdateModelCheck)
if(NOT WIN32)
  return()
endif()
add_compile_definitions(NOMINMAX UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8)
endif()
add_executable(RushUpdateFixture fixture-launcher.cpp)
add_executable(RushEditionUpdateCheck editions-check.cpp)
add_executable(PortUpdateAgent WIN32 ../windows_native/port-update-agent.cpp)
add_executable(RushUpdateAgentCheck ../windows_native/rush-update-agent-check.cpp)
add_executable(RushNativeUpdateUICheck native-ui-check.cpp)
add_executable(RushUpdateNetworkCheck network-check.cpp)
foreach(target RushEditionUpdateCheck PortUpdateAgent RushUpdateAgentCheck RushNativeUpdateUICheck RushUpdateNetworkCheck)
  target_include_directories(${target} PRIVATE "${RUSH_JSON_INCLUDE_DIR}")
  target_link_libraries(${target} PRIVATE winhttp bcrypt shell32 user32 wintrust crypt32 ole32 uuid comctl32 gdi32)
endforeach()
if(MINGW)
  add_link_options(-static)
  foreach(target RushUpdateFixture RushEditionUpdateCheck PortUpdateAgent RushUpdateAgentCheck RushNativeUpdateUICheck RushUpdateNetworkCheck)
    target_link_options(${target} PRIVATE -municode -static)
  endforeach()
endif()
add_test(NAME edition_update_transactions COMMAND RushEditionUpdateCheck $<TARGET_FILE:RushUpdateFixture>)
add_test(NAME stopped_game_and_preservation COMMAND RushUpdateAgentCheck)
add_test(NAME native_updates_panel COMMAND RushNativeUpdateUICheck)

option(RUSH_TEST_LIVE_UPDATES "Check the public GitHub update service without executing downloads" OFF)
if(RUSH_TEST_LIVE_UPDATES)
  add_test(NAME live_update_transport COMMAND RushUpdateNetworkCheck)
  set_tests_properties(live_update_transport PROPERTIES TIMEOUT 180)
endif()
