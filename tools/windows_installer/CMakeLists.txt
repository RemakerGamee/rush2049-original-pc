cmake_minimum_required(VERSION 3.20)
project(RushOriginalInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(RUSH_BUILD_SETUP)
  add_executable(Rush2049OriginalSetup WIN32 app.cpp "${RUSH_RESOURCE_DIR}/setup.rc")
  target_compile_definitions(Rush2049OriginalSetup PRIVATE RUSH_SETUP)
  set(TARGET Rush2049OriginalSetup)
else()
  add_executable(Rush2049Original WIN32 app.cpp "${RUSH_RESOURCE_DIR}/icon.rc")
  set(TARGET Rush2049Original)
endif()
target_include_directories(${TARGET} PRIVATE "${RUSH_RESOURCE_DIR}")
target_compile_options(${TARGET} PRIVATE /utf-8)
target_link_libraries(${TARGET} PRIVATE bcrypt comdlg32 comctl32 shell32 ole32 uuid user32)
target_link_options(${TARGET} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/application.manifest")

if(RUSH_PACKAGE_VERSION)
  target_compile_definitions(${TARGET} PRIVATE RUSH_PACKAGED_VERSION="${RUSH_PACKAGE_VERSION}")
endif()
